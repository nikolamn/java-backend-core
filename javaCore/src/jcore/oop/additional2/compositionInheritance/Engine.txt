Inheritance
class Engine {
    void start() { System.out.println("Engine started"); }
}

class Car extends Engine {
    void drive() { System.out.println("Car driving"); }
}

Composition
class Engine {
    void start() { System.out.println("Engine started"); }
}

class Car {

    private Engine engine;

    Car(Engine engine) { this.engine = engine; }

    void drive() {
        engine.start();
        System.out.println("Car driving");
    }
}
