# Lazy loading users
curl -X GET -i http://localhost:8080/api/demo/users-lazy \
-H "Accept: application/json"

# Eager loading users
curl -X GET -i http://localhost:8080/api/demo/users-eager \
-H "Accept: application/json"

-H "Accept: text/plain"
